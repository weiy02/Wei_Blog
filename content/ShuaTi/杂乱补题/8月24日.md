#pragma GCC optimize("O3")
#pragma GCC optimize("unroll-loops")
#include<bits/stdc++.h>
using namespace std;
using ll = long long;
const ll N=1e5+5,M=2e6+5,mod=998244353,inf=1LL<<60;
const long double pi=acos(-1);
int fa[N];
int find(int x){
    if(fa[x]!=x) fa[x] = find(fa[x]);
    return fa[x];
}
void unite(int x, int y){
    x = find(x);
    y = find(y);
    if(x!=y) fa[y] = x;
}
void init(int n){
    for(int i = 1;i<=n;i++){
        fa[i] = i;
    }
}
void solve(){
    int n,m;
    while(cin >> n >> m){
        init(n);
        while(m--){
            int a,b;
            cin >> a >> b;
            unite(a,b);
        }
        if(find(1)==find(n)) cout << "Yes" << endl;
        else cout << "No" << endl;
    }
}

signed main(){
    ios::sync_with_stdio(0),cin.tie(0),cout.tie(0);
    cout << fixed << setprecision(10);
    int T=1;
    //cin >> T;
    while(T--) solve();
    return 0;
}
