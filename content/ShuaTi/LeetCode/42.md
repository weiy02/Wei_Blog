class Solution {
public:
    int trap(vector<int>& height) {
        int res = 0;
        int i = 0, j = height.size()-1;
        res += (j-i-2) * min(height[i],height[j]);
        while(i<j){
            if(height[i]<=height[j]){
                i++;
                res -= height[i];
                if(height[i]>min(height[i-1],height[j]))
                    res += (j-i-2) * abs(height[i]-height[j]);
            }else if(height[i]>height[j]){
                j--;
                res -= height[j];
                if(height[j]>min(height[i],height[j+1]))
                    res += (j-i-2) * abs(height[i]-height[j]);
            }
        }
        return res;
    }
};
